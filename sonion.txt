local function IsPlayerCharacter(obj)
    local model = obj:FindFirstAncestorOfClass("Model")
    return model and Players:GetPlayerFromCharacter(model) ~= nil
end

local function XrayKeep(part)
    if XRAY_KEEP[part.Name] then return true end
    if IsPlayerCharacter(part) then return true end
    local myChar = LocalPlayer.Character
    if myChar and part:IsDescendantOf(myChar) then return true end
    local cur = part
    while cur and cur ~= Workspace do
        if XRAY_KEEP_FOLDERS[cur.Name] then return true end
        if cur.Name == "Ground" or cur.Name == "Carpet" or cur.Name == "Terrain" then
            return true
        end
        cur = cur.Parent
    end
    return false
end

local function IsXrayWanted(part)
    local cur = part
    while cur and cur ~= Workspace do
        if XRAY_WANTED[cur.Name] then return true end
        cur = cur.Parent
    end
    return false
end

local function Save(store, obj, prop, value)
    local s = Core.Opt.Saved[store]
    if not s then s = {} Core.Opt.Saved[store] = s end
    if not s[obj] then s[obj] = {} end
    if s[obj][prop] == nil then s[obj][prop] = obj[prop] end
    obj[prop] = value
end

local function RestoreAll(store)
    local s = Core.Opt.Saved[store]
    if not s then return end
    for obj, props in s do
        if obj and obj.Parent then
            for prop, value in props do
                pcall(function() obj[prop] = value end)
            end
        end
    end
    Core.Opt.Saved[store] = nil
end

local function AddConn(store, c)
    local t = Core.Opt.Trash[store]
    if not t then t = {} Core.Opt.Trash[store] = t end
    table.insert(t, c)
end

local function CleanConns(store)
    local t = Core.Opt.Trash[store]
    if not t then return end
    for _, c in t do pcall(function() c:Disconnect() end) end
    Core.Opt.Trash[store] = nil
end

    AddConn("particles", Workspace.DescendantAdded:Connect(function(obj)
        if particleClasses[obj.ClassName] then
            task.defer(function() if obj.Parent then obj:Destroy() end end)
        end
    end))
    for _, d in Workspace:GetDescendants() do
        if particleClasses[d.ClassName] then d:Destroy() end
    end

local function Flatten(part)
    if not part:IsA("BasePart") then return end
    if part:FindFirstChildWhichIsA("ProximityPrompt", true) then return end
    if part.Material ~= Enum.Material.SmoothPlastic then
        Save("materials", part, "Material", Enum.Material.SmoothPlastic)
    end
    Save("materials", part, "Reflectance", 0)
    Save("materials", part, "CastShadow", false)
end
local function MaterialsOn()
    AddConn("materials", Workspace.DescendantAdded:Connect(function(obj)
        if obj:IsA("BasePart") then task.defer(Flatten, obj) end
    end))
    for _, d in Workspace:GetDescendants() do Flatten(d) end
end
local function MaterialsOff() CleanConns("materials") RestoreAll("materials") end

local function QuietLighting(obj)
    if obj:IsA("PostEffect") then
        Save("lighting", obj, "Enabled", false)
    elseif obj:IsA("Sky") or obj:IsA("Atmosphere") then
        Save("lighting", obj, "Parent", nil)
    end
end
local function LightingOn()
    Save("lighting", Lighting, "GlobalShadows", false)
    Save("lighting", Lighting, "FogEnd", 1e9)
    Save("lighting", Lighting, "FogStart", 0)
    Save("lighting", Lighting, "Brightness", 1)
    Save("lighting", Lighting, "Ambient", Color3.new(1,1,1))
    Save("lighting", Lighting, "OutdoorAmbient", Color3.new(1,1,1))
    Save("lighting", Lighting, "EnvironmentDiffuseScale", 0)
    Save("lighting", Lighting, "EnvironmentSpecularScale", 0)
    pcall(function() Save("lighting", Lighting, "Technology", Enum.Technology.Compatibility) end)
    for _, obj in Lighting:GetChildren() do QuietLighting(obj) end
    AddConn("lighting", Lighting.ChildAdded:Connect(function(obj)
        task.defer(QuietLighting, obj)
    end))
end
local function LightingOff() CleanConns("lighting") RestoreAll("lighting") end

local function FreezeAnim(anim)
    if Core.FrozenAnims[anim] then return end
    Core.FrozenAnims[anim] = {Speed = anim.Speed, Weight = anim.WeightCurrent}
    pcall(function()
        anim:AdjustSpeed(0)
        anim:AdjustWeight(0)
    end)
end

local function ThawAnim(anim)
    local data = Core.FrozenAnims[anim]
    if not data then return end
    pcall(function()
        anim:AdjustSpeed(data.Speed)
        anim:AdjustWeight(data.Weight)
    end)
    Core.FrozenAnims[anim] = nil
end

local function IsBrainrotAnimator(animator)
    local cur = animator
    while cur and cur ~= Workspace do
        if cur.Name == "RenderedMovingAnimals" then return true end
        if cur.Name == "Animals" then return true end
        cur = cur.Parent
    end
    return false
end

local function MuteAnimator(animator)
    if not animator:IsA("Animator") then return end
    if not IsBrainrotAnimator(animator) then return end
    for _, track in animator:GetPlayingAnimationTracks() do
        FreezeAnim(track)
    end
    local conn = animator.AnimationPlayed:Connect(FreezeAnim)
    table.insert(Core.FrozenConns, conn)
end

local function AnimationsOn()
    for _, d in ipairs(Workspace:GetDescendants()) do
        if d:IsA("Animator") then MuteAnimator(d) end
    end
    local watchConn = Workspace.DescendantAdded:Connect(function(obj)
        if obj:IsA("Animator") then
            task.defer(MuteAnimator, obj)
        end
    end)
    table.insert(Core.FrozenConns, watchConn)
end

local function AnimationsOff()
    for _, conn in ipairs(Core.FrozenConns) do
        pcall(function() conn:Disconnect() end)
    end
    table.clear(Core.FrozenConns)
    for anim in Core.FrozenAnims do
        ThawAnim(anim)
    end
    table.clear(Core.FrozenAnims)
end

local function Paint(part)
    if XrayKeep(part) then return end
    if Core.Painted[part] then return end
    local texture = nil
    if part:IsA("MeshPart") and part.TextureID ~= "" then
        texture = part.TextureID
    end
    local data = {Color = part.Color, Texture = texture, Tinted = false}
    Core.Painted[part] = data
    part.LocalTransparencyModifier = Core.STYLE_XRAY.Amount
    part.Color = Core.STYLE_XRAY.Color
    if texture then part.TextureID = "" end
    data.Tinted = true
end

local function Unpaint(part)
    local data = Core.Painted[part]
    if not data then return end
    if part.Parent then
        pcall(function()
            part.LocalTransparencyModifier = 0
            part.Color = data.Color
            if data.Texture then part.TextureID = data.Texture end
        end)
    end
    Core.Painted[part] = nil
end

local function HideGui(guiObj)
    if guiObj:IsA("SurfaceGui") or guiObj:IsA("BillboardGui") then
        if not Core.Opt.Saved["guis"] then Core.Opt.Saved["guis"] = {} end
        if not Core.Opt.Saved["guis"][guiObj] then
            Core.Opt.Saved["guis"][guiObj] = {Enabled = guiObj.Enabled}
        end
        guiObj.Enabled = false
    end
end

local function RestoreGui(guiObj)
    local s = Core.Opt.Saved["guis"]
    if not s then return end
    local data = s[guiObj]
    if not data then return end
    pcall(function() guiObj.Enabled = data.Enabled end)
    s[guiObj] = nil
end

local function HideGuiTree(parent)
    for _, child in ipairs(parent:GetDescendants()) do
        if child:IsA("SurfaceGui") or child:IsA("BillboardGui") then
            HideGui(child)
        end
    end
end

local function XrayAll()
    for _, part in ipairs(Workspace:GetDescendants()) do
        if part:IsA("BasePart") and part.Parent then
            if IsXrayWanted(part) and not XrayKeep(part) then
                Paint(part)
                HideGuiTree(part)
            end
        elseif part:IsA("SurfaceGui") or part:IsA("BillboardGui") then
            local adornee = part.Adornee
            local target = adornee or part.Parent
            if target then
                local shouldHide = false
                if target:IsA("BasePart") and IsXrayWanted(target) and not XrayKeep(target) then
                    shouldHide = true
                end
                if not shouldHide then
                    local cur = target
                    while cur and cur ~= Workspace do
                        if XRAY_WANTED[cur.Name] then
                            shouldHide = true
                            break
                        end
                        cur = cur.Parent
                    end
                end
                if shouldHide then HideGui(part) end
            end
        end
    end
end

local function UnxrayAll()
    for part in Core.Painted do
        Unpaint(part)
    end
    table.clear(Core.Painted)
    local s = Core.Opt.Saved["guis"]
    if s then
        for guiObj in s do
            RestoreGui(guiObj)
        end
        Core.Opt.Saved["guis"] = nil
    end
end

Core.XrayAll = XrayAll

-- CARPET
local carpetHidden = false
local carpetPart = nil
local carpetConn = nil

local function HideCarpet()
    if carpetHidden then return end
    local map = Workspace:FindFirstChild("Map")
    local carpet = map and map:FindFirstChild("Carpet")
    if not carpet then
        warn("[RDOZ] Map.Carpet not found")
        return
    end
    carpetHidden = true
    carpetPart = carpet
    local function Force()
        if carpet.Parent then
            if carpet.Transparency ~= 1 then carpet.Transparency = 1 end
            if carpet.LocalTransparencyModifier ~= 1 then carpet.LocalTransparencyModifier = 1 end
            if carpet.CanCollide then carpet.CanCollide = false end
        end
    end
    Force()
    carpetConn = RunService.Heartbeat:Connect(Force)
    warn("[RDOZ] Carpet hidden")
end

local function RestoreCarpet()
    if not carpetHidden then return end
    carpetHidden = false
    if carpetConn then
        carpetConn:Disconnect()
        carpetConn = nil
    end
    if carpetPart and carpetPart.Parent then
        carpetPart.Transparency = 0
        carpetPart.LocalTransparencyModifier = 0
        carpetPart.CanCollide = true
    end
    carpetPart = nil
    warn("[RDOZ] Carpet restored")
end

Core.HideCarpet = HideCarpet
Core.RestoreCarpet = RestoreCarpet

    Core.Opt.On = true
    pcall(ParticlesOn)
    pcall(EffectsOn)
    pcall(ViewportsOn)
    pcall(LightsOn)
    pcall(SoundsOn)
    pcall(MaterialsOn)
    pcall(LightingOn)
    pcall(TerrainOn)
    pcall(QualityOn)
    pcall(OverheadsOn)
    pcall(TexturesOn)
    pcall(AccessoriesOn)
    pcall(AnimationsOn)
    pcall(HideCarpet)
    Core.XrayOn = true
    task.spawn(XrayAll)
    Core.FreezeOn = true
end

local function AllOff()
    Core.Opt.On = false
    pcall(ParticlesOff)
    pcall(EffectsOff)
    pcall(ViewportsOff)
    pcall(LightsOff)
    pcall(SoundsOff)
    pcall(MaterialsOff)
    pcall(LightingOff)
    pcall(TerrainOff)
    pcall(QualityOff)
    pcall(OverheadsOff)
    pcall(TexturesOff)
    pcall(AccessoriesOff)
    pcall(AnimationsOff)
    pcall(RestoreCarpet)
    Core.XrayOn = false
    UnxrayAll()
    Core.FreezeOn = false
end
